        const FormData = require('form-data')
        const axios    = require('axios')

        async function criarJob(buffers, p) {
            const form = new FormData()
            form.append('image', buffers[0], 'image1.jpg')
            if (buffers[1]) form.append('image2', buffers[1], 'image2.jpg')
            form.append('prompt', p)

            const r = await axios.post('https://zone.api.br/api/v2/nano-banana?apikey=freekey', form, {
                headers: form.getHeaders(),
                maxBodyLength: Infinity,
                maxContentLength: Infinity,
                timeout: 60000,
            })
            if (!r.data?.status || !r.data?.job_id) throw new Error('Falha ao criar o job de edição')
            return r.data.job_id
        }

        async function checarJob(jobId) {
            const r = await axios.get('https://zone.api.br/api/v2/nano-banana/status', {
                params: { job_id: jobId },
                timeout: 15000,
            })
            return r.data
